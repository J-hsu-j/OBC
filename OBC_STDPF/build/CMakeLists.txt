#
# function ADD_OPTION:
# The function performs the following actions:
#     - adds a standard CMake option with name <option>, a description specified
#       with the <description> parameter and a default value given by <default_value>
#       parameter
#     - appends the '-D<option>' preprocessor-define to the list referenced by the <cdef_list> parameter
#     - generates a '#define <option>' line in the ${BUILD_OPT_H} file if the option is turned on
#
# Example:
#     ADD_OPTION(BEACONS_ENABLED "Specifies whether beacons sending is active or not" ON ROOT_COMPILE_DEFS)
#
#     The BEACONS_ENABLED CMake option will be added and accessible via the cmake-gui. It will be turned ON by
#     default, thus our build_options.h file will contain the following line:
#         '#define BEACONS_ENABLED'.
#     Finally the ROOT_COMPILE_DEFS list will be appended with '-DBEACONS_ENABLED' compiler option and this preprocessor
#     define will be propagated to all files in the build, making it possible to use conditional compilation without including
#     any specific header:
#
#     #if defined(BEACONS_ENABLED)
#         <Beacons-specific code here>
#     #endif    // #if defined(BEACONS_ENABLED)
#
function(ADD_OPTION option description default_value cdef_list)
    option(${option} ${description} ${default_value})

    if (${default_value})
        list(APPEND templist
             ${${cdef_list}}
             ${option}
        )
        set(${cdef_list} ${templist} PARENT_SCOPE)

        file(APPEND ${BUILD_OPT_H} "#define ${option}\n")
    else ()
        file(APPEND ${BUILD_OPT_H} "// #define ${option}\n")
    endif ()
endfunction()

#
# function TRY_LINK_EXISTING_LIB:
# Attempts to link a prebuilt static library to a specified target. If the library does not exist at the path
# generated by the GENERATE_LIB_PATH function, it assumes the source code and a CMakeLists.txt for building the
# library are available at a specified path (LIB_SRC_PATH). The function then builds the library from source at the
# specified output directory (LIB_BIN_PATH) and links it to the target. This function is useful where
# prebuilt libraries are preferred, but a fallback to source compilation is necessary if the library is not found.
#
# Parameters:
#   <TARGET_NAME> (string): The name of the target to which the library should be linked.
#   <PREBUILT_LIB_NAME> (string): The name of the prebuilt library file (including extension, e.g., libexample.a).
#   <LIB_SRC_PATH> (string): The absolute directory path where the CMakeLists.txt for building the library from source is located.
#   <LIB_BIN_PATH> (string): The absolute directory path where the built library should be output if it is built from source.
#   <LIB_TARGET_NAME> (string): The target name defined in the CMakeLists.txt for the library when built from source.
#
# Example:
#    TRY_LINK_EXISTING_LIB(
#       "MyApplication"                 # TARGET_NAME
#       "libexternal.a"                 # PREBUILT_LIB_NAME
#       "${PROJECT_SOURCE_DIR}/module"  # LIB_SRC_PATH
#       "${PROJECT_BINARY_DIR}/module"  # LIB_BIN_PATH
#       "external")                     # LIB_TARGET_NAME
#
#    This example attempts to link 'libexternal.a' to the 'MyApplication' target. If the library
#    does not exist, it builds it from source located in '${PROJECT_SOURCE_DIR}/module' and outputs
#    the build in '${PROJECT_BINARY_DIR}/module', linking the resulting 'external' target to 'MyApplication'.
function(TRY_LINK_EXISTING_LIB TARGET_NAME PREBUILT_LIB_NAME LIB_SRC_PATH LIB_BIN_PATH LIB_TARGET_NAME)
    # Generate the path to the expected library location using the build configuration
    GENERATE_LIB_PATH(${PREBUILT_LIB_NAME} FULL_LIB_PATH)

    message(CHECK_START "Looking for pre-built library '${FULL_LIB_PATH}'")
    if (EXISTS ${FULL_LIB_PATH})
        message(CHECK_PASS "Found - will perform direct linkage.")
        target_link_libraries(${TARGET_NAME} PUBLIC "${FULL_LIB_PATH}")
    else ()
        message(CHECK_FAIL "Not found - will attempt to build from source.")
        add_subdirectory(${LIB_SRC_PATH} ${LIB_BIN_PATH})
        target_link_libraries(${TARGET_NAME} PUBLIC ${LIB_TARGET_NAME})
        message("Library built from source and linked.")
    endif ()
endfunction()

# Add the espf subdirectory where all the sources are.
# IMPORTANT to be done after defining all the global compile and linker options and definitions
add_subdirectory(${PROJECT_ROOT} espf)

# Build the OBC target
add_executable(${OBC_TARGET})

# IMPORTANT build after including the espf subdirectory since it populates the sources, includes and libraries
target_sources(${OBC_TARGET} PRIVATE
    ${OBC_sources}
)
target_include_directories(${OBC_TARGET} PUBLIC
    ${OBC_includes}
)
target_link_libraries(${OBC_TARGET} PUBLIC
    ${OBC_libraries}
)


# This line specifies the linker definitions file used for configurations without a
# bootloader (a.k.a. No-boot configurations). These can be flashed stand-alone on the OBC
# without the need for a pre-flashed Endurosat Bootloader image to run them.
set(NOBOOT_LINKER_FILE
    "-T${PROJECT_ROOT}/arch/stm32h753iit/linker/STM32H753IITx_FLASH_No_Bootloader.ld"
)

# This line specifies the linker definitions file for the configurations which will be used
# alongside the Endurosat Bootloader image.
set(APP_LINKER_FILE
    "-T${PROJECT_ROOT}/arch/stm32h753iit/linker//STM32H753IITx_FLASH.ld"
)

# OBC Public definitions
target_compile_definitions(${OBC_TARGET} PUBLIC
    "-D__packed=__attribute__\(\(__packed__\)\)"
    "-D__weak=__attribute__\(\(weak\)\)"
)
# Private defininitions (so updating the version does not trigger the build of every library)
target_compile_definitions(${OBC_TARGET} PRIVATE
    "-D${CMAKE_PROJECT_NAME}_VERSION_MAJOR=${${CMAKE_PROJECT_NAME}_VERSION_MAJOR}"
    "-D${CMAKE_PROJECT_NAME}_VERSION_MINOR=${${CMAKE_PROJECT_NAME}_VERSION_MINOR}"
    "-D${CMAKE_PROJECT_NAME}_VERSION_PATCH=${${CMAKE_PROJECT_NAME}_VERSION_PATCH}"
    "-D${CMAKE_PROJECT_NAME}_VERSION_LABEL=${${CMAKE_PROJECT_NAME}_VERSION_LABEL_TRIMMED}"
)

# The following section specifies the OBC SDK executable linker options. As you can see, some of
# these options depend on the build configuration used.
target_link_options(${OBC_TARGET} PUBLIC
    $<$<CONFIG:noboot_debug>:${NOBOOT_LINKER_FILE}>
    $<$<CONFIG:noboot_release>:${NOBOOT_LINKER_FILE}>
    $<$<CONFIG:debug>:${APP_LINKER_FILE}>
    $<$<CONFIG:Release>:${APP_LINKER_FILE}>
    -Wl,-Map=$<CONFIG>/${OBC_TARGET}_$<CONFIG>.map
)

# Module ID must always be built since it contains versioning information and the timestamp of the build
set(MODULE_ID_PATH "${PROJECT_ROOT}/core/lib/module_id/src/module_id.c")
add_custom_command(
    TARGET ${OBC_TARGET} PRE_BUILD COMMAND ${CMAKE_COMMAND} -E touch
    "${MODULE_ID_PATH}"
    VERBATIM
)
# This custom command doesn't actually generate any output but it triggers a rebuild.
add_custom_command(
    OUTPUT "${MODULE_ID_PATH}"
    COMMAND ${CMAKE_COMMAND} -E touch "${MODULE_ID_PATH}"
    VERBATIM
)

# This custom command generates some build artifacts after linking the final executable image.
# It prints the build size summary and generates the linker map file, the compiler listing file and
# a .bin file ready for flashing via the STM32Programmer tool. Finally, the command performs some clean-up
# activities for any FIDL-generated content.
add_custom_command(
    TARGET ${OBC_TARGET} POST_BUILD
    COMMAND arm-none-eabi-size "$<TARGET_FILE:${OBC_TARGET}>"
    COMMAND arm-none-eabi-objcopy -O binary "$<TARGET_FILE:${OBC_TARGET}>" "$<CONFIG>/${OBC_TARGET}_$<CONFIG>.bin"
    COMMAND arm-none-eabi-objdump -h -S "$<TARGET_FILE:${OBC_TARGET}>" > "$<CONFIG>/${OBC_TARGET}_$<CONFIG>.list"
    COMMAND ${MACARON_CMD} --clean-gen-content=${PROJECT_ROOT}
    VERBATIM
)
